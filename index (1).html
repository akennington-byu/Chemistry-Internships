        get filteredPrograms() {

          return this.programs.filter(item => {

            const q = this.searchQuery.toLowerCase();

            const matchesSearch = q === '' || 

              item.name.toLowerCase().includes(q) ||

              item.organization.toLowerCase().includes(q) ||

              item.description.toLowerCase().includes(q);


            const matchesCategory = this.selectedCategory === 'All' || item.category === this.selectedCategory;

            const matchesYear = this.selectedYear === 'All' || item.eligibleYears.includes(this.selectedYear);

            const matchesStatus = this.selectedStatus === 'All' || item.status === this.selectedStatus;

            const matchesIntl = !this.intlEligibleOnly || item.intlEligible;

            const matchesSaved = !this.showSavedOnly || this.savedIds.includes(item.id);


            return matchesSearch && matchesCategory && matchesYear && matchesStatus && matchesIntl && matchesSaved;

          });

          }

          localStorage.setItem('savedOpportunities', JSON.stringify(this.savedIds));

        isSaved(id) {

          return this.savedIds.includes(id);

        },


        resetFilters() {

          this.searchQuery = '';

          this.selectedCategory = 'All';

          this.selectedYear = 'All';

          this.selectedStatus = 'All';

          this.intlEligibleOnly = false;

          this.showSavedOnly = false;

        },


        exportShortlistCSV() {

          const savedItems = this.programs.filter(p => this.savedIds.includes(p.id));

          const headers = ['Program Name', 'Organization', 'Category', 'Typical Deadline', 'Funding', 'URL'];

          const rows = savedItems.map(p => [

            `"${p.name}"`, `"${p.organization}"`, `"${p.category}"`, `"${p.typicalDeadline}"`, `"${p.funding}"`, `"${p.url}"`

          ]);

          const csvContent = "data:text/csv;charset=utf-8," + [headers.join(','), ...rows.map(e => e.join(','))].join('\n');

          const encodedUri = encodeURI(csvContent);

          const link = document.createElement("a");

          link.setAttribute("href", encodedUri);

          link.setAttribute("download", "my_internship_shortlist.csv");

          document.body.appendChild(link);

          link.click();

          document.body.removeChild(link);

        }

      };
